#!/bin/bash

LOG_FILE="$HOME/logs/execucao_batch.log"
INPUT_DIR="$HOME/input"
OUTPUT_DIR="$HOME/output"

echo "[$(date '+%Y-%m-%d %H:%M:%S')] [INFO] Iniciando ciclo de processamento Batch..." >> "$LOG_FILE"

# Verifica se existem arquivos .txt na pasta input
if [ -n "$(ls -A "$INPUT_DIR"/*.txt 2>/dev/null)" ]; then
    for arquivo in "$INPUT_DIR"/*.txt; do
        nome_arquivo=$(basename "$arquivo")
        echo "[$(date '+%Y-%m-%d %H:%M:%S')] [INFO] Processando o arquivo: $nome_arquivo" >> "$LOG_FILE"
        
        sleep 3
        
        mv "$arquivo" "$OUTPUT_DIR/"
        echo "[$(date '+%Y-%m-%d %H:%M:%S')] [SUCCESS] Arquivo $nome_arquivo processado com sucesso." >> "$LOG_FILE"
    done
else
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [WARNING] Nenhum arquivo encontrado para processamento nesta janela." >> "$LOG_FILE"
fi

echo "[$(date '+%Y-%m-%d %H:%M:%S')] [INFO] Ciclo Batch finalizado." >> "$LOG_FILE"
